def resolve_paths(args, *, arg_names=..., constant=None, clipboard=True, ...):
    cli_values = _collect_arg_values(args, arg_names)
    if cli_values:
        return ResolvedPaths(paths=..., origin="args")

    # NEW: stdin support (second priority)
    stdin_values = _read_paths_from_stdin()
    if stdin_values:
        return ResolvedPaths(paths=..., origin="stdin")

    if clipboard:
        clip = paths_from_clipboard()
        if clip:
            return ResolvedPaths(paths=..., origin="clipboard")

    const_paths = _flatten_paths(constant)
    if const_paths and _exists(const_paths):
        return ResolvedPaths(paths=..., origin="constant")

    return ResolvedPaths(paths=(Path.cwd(),), origin="cwd")
stdin + stdout Handling
Text content (for writers):
Priority: --text arg > sys.stdin.read() > clipboard text
Paths (for both readers & writers):
Priority: args > stdin > clipboard > constant > cwd
Stdout for writers:
Only when --stdout (or equivalent) is passed in arguments.
Otherwise always writes to a file using choose_output_file().
